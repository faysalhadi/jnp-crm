const loadFollowUp = async () => {
  const { data } = await supabase
    .from("follow_ups")
    .select("*")
    .eq("customer_id", activeCustomerId)
    .eq("status", "pending")
    .order("due_at", { ascending: true })
    .limit(1)
    .maybeSingle();
  setFollowUp(data || null);
};
